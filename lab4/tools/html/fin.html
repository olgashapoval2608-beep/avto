SELECT book.id, book.title, author.name AS author, book.publication_year, author.address AS author_address, CONCAT(publisher.city, ', ', publisher.country) AS publisher_address, book.price, bookstore.name AS bookstore FROM book, author, publisher, bookstore WHERE book.id_author = author.id_author AND book.id_publisher = publisher.id_publisher AND book.id_bookstore = bookstore.id_bookstore ORDER BY book.id;
